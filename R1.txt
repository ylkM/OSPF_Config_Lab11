! =====================================================
! R1 Configuration
! Role: ASBR - connects OSPF domain to ISP (203.0.113.0/30)
! =====================================================
enable
configure terminal

hostname R1

no ip domain-lookup

! --- Interfaces ---
interface GigabitEthernet3/0
 description WAN link to ISPR1 - NOT in OSPF
 ip address 203.0.113.1 255.255.255.252
 no shutdown
 exit

interface GigabitEthernet0/0
 description LAN link to R2
 ip address 10.0.12.1 255.255.255.252
 no shutdown
 exit

interface FastEthernet1/0
 description LAN link to R3
 ip address 10.0.13.1 255.255.255.252
 no shutdown
 exit

interface Loopback0
 description OSPF Router ID / test loopback
 ip address 1.1.1.1 255.255.255.255
 exit

! --- Static default route out to the Internet ---
ip route 0.0.0.0 0.0.0.0 203.0.113.2

! --- OSPF Process ---
router ospf 1
 router-id 1.1.1.1
 network 10.0.12.0 0.0.0.3 area 0
 network 10.0.13.0 0.0.0.3 area 0
 network 1.1.1.1 0.0.0.0 area 0
 passive-interface Loopback0
 ! R1 is the ASBR - originate a default route into OSPF
 default-information originate
 exit

line con 0
 logging synchronous
 exit

end
write memory
